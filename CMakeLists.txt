cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(CursorBrightness VERSION 1.2.1 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "Cursor Brightness requires Windows")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(CursorBrightness WIN32 app/main.cpp app/resources.rc)
target_include_directories(CursorBrightness PRIVATE app assets)
target_compile_definitions(CursorBrightness PRIVATE UNICODE _UNICODE NOMINMAX WINVER=0x0A00 _WIN32_WINNT=0x0A00 NTDDI_VERSION=0x0A000008)
target_link_libraries(CursorBrightness PRIVATE dxva2 ole32 oleaut32 wbemuuid uuid comctl32 shell32 gdi32 dwmapi hid wtsapi32 taskschd advapi32)
if(MSVC)
  target_compile_options(CursorBrightness PRIVATE /W4 /utf-8 /EHsc)
  target_link_options(CursorBrightness PRIVATE /MANIFEST:NO)
  target_link_libraries(CursorBrightness PRIVATE comsuppw)
else()
  target_compile_options(CursorBrightness PRIVATE -Wall -Wextra)
  target_link_options(CursorBrightness PRIVATE -static -municode)
endif()
enable_testing()
add_executable(brightness-core-tests tests/brightness-core.cpp)
if(MSVC)
  target_compile_options(brightness-core-tests PRIVATE /UNDEBUG /utf-8)
else()
  target_compile_options(brightness-core-tests PRIVATE -UNDEBUG)
  target_link_options(brightness-core-tests PRIVATE -static)
endif()
add_test(NAME brightness-core COMMAND brightness-core-tests)
add_executable(native-popup-state-tests tests/native-popup-state.cpp)
target_compile_definitions(native-popup-state-tests PRIVATE UNICODE _UNICODE NOMINMAX)
if(MSVC)
  target_compile_options(native-popup-state-tests PRIVATE /UNDEBUG /utf-8)
else()
  target_compile_options(native-popup-state-tests PRIVATE -UNDEBUG)
  target_link_options(native-popup-state-tests PRIVATE -static)
endif()
add_test(NAME native-popup-state COMMAND native-popup-state-tests)
install(TARGETS CursorBrightness RUNTIME DESTINATION .)
install(FILES LICENSE README.md README.ru.md THIRD_PARTY_NOTICES.md THIRD_PARTY_NOTICES.ru.md DESTINATION .)
